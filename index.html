/* 1. Pure Character Stack in C */
#define MAX 100

typedef struct {
    char items[MAX];  /* Each element of the stack is a character */
    int top;
} Stack;

Stack undoStack, redoStack;
char currentText[MAX + 1] = "";

/* 2. Push & Pop Character Functions */
int push(Stack *s, char c) {
    if (s->top == MAX - 1) return 0; /* Stack Overflow */
    s->top++;
    s->items[s->top] = c;
    return 1;
}

char pop(Stack *s) {
    if (s->top == -1) return '\0';   /* Stack Underflow */
    char c = s->items[s->top];
    s->top--;
    return c;
}
